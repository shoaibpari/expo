// --- FIREBASE IMPORTS (Naye Tareeqay Se) ---
// Note: Instead of 'firebase/app', we use the direct modular paths.
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, onSnapshot, query, addDoc, doc, deleteDoc, runTransaction } from 'firebase/firestore';

// --- GLOBAL FIREBASE CONFIGURATION (MANDATORY USE) ---
const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
const firebaseConfig = JSON.parse(typeof __firebase_config !== 'undefined' ? __firebase_config : '{}');

let app;
let db;
let auth;

// Global Firebase initialization and authentication status
const useFirebaseSetup = () => {
  const [userId, setUserId] = useState(null);
  const [isAuthReady, setIsAuthReady] = useState(false);
  const [dbInstance, setDbInstance] = useState(null);
  const [authInstance, setAuthInstance] = useState(null);
  const [initError, setInitError] = useState(null);

  useEffect(() => {
    let unsubscribe;
    
    try {
      // Initialize only once
      if (!app) {
        app = initializeApp(firebaseConfig);
      }
      if (!dbInstance) {
        setDbInstance(getFirestore(app));
      }
      if (!authInstance) {
        const currentAuth = getAuth(app);
        setAuthInstance(currentAuth);
      
        // --- Authentication Logic ---
        const authenticate = async (auth) => {
          try {
            if (typeof __initial_auth_token !== 'undefined') {
              await signInWithCustomToken(auth, __initial_auth_token);
            } else {
              await signInAnonymously(auth);
            }
          } catch (error) {
            console.error("Firebase Auth Error:", error);
          }
        };

        // Listen for authentication state changes
        unsubscribe = onAuthStateChanged(currentAuth, (user) => {
          if (user) {
            setUserId(user.uid);
          } else {
            // Fallback for anonymous sign-in if token was missing or failed
            if (!currentAuth.currentUser) authenticate(currentAuth); 
          }
          setIsAuthReady(true);
        });

        // Initial sign-in attempt
        authenticate(currentAuth);
      }
    } catch (error) {
      console.error("Firebase initialization error:", error);
      setInitError("Firebase Init Failed. Check package.json & internet.");
      setIsAuthReady(true);
    }
    
    return () => {
      if (unsubscribe) unsubscribe();
    };
  }, []); // Run only on mount

  return { db: dbInstance, auth: authInstance, userId, isAuthReady, initError };
};


// --- GLOBAL DATA FETCHING HOOK ---
const useOliveData = (db, userId, isAuthReady) => {
  const [products, setProducts] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Wait for DB and Auth to be ready
    if (!db || !userId || !isAuthReady) return;

    // Path to the public collection for shared products
    const productsPath = `/artifacts/${appId}/public/data/products`;
    const q = query(collection(db, productsPath));
    
    // Real-time listener for products
    const unsubscribe = onSnapshot(q, (snapshot) => {
      const productList = [];
      snapshot.forEach((doc) => {
        const data = doc.data();
        // Ensure price is a number for calculations
        const price = typeof data.price === 'string' ? parseFloat(data.price.replace('Rs ', '')) : data.price;
        
        productList.push({
          id: doc.id,
          ...data,
          price: price || 0 // Use 0 if price parsing fails
        });
      });
      setProducts(productList);
      setLoading(false);
    }, (error) => {
      console.error("Error fetching products:", error);
      setLoading(false);
    });

    return () => unsubscribe();
  }, [db, userId, isAuthReady]);

  return { products, loading };
};

// --- MAIN APP (Navigation & State Logic) ---
export default function App() {
  const [activeTab, setActiveTab] = useState('Home');
  const [cart, setCart] = useState([]); // Temporary local cart state
  
  // Custom hook to handle Firebase authentication & setup
  const { db, userId, isAuthReady, initError } = useFirebaseSetup();
  // Custom hook to fetch products from Firestore
  const { products, loading: productsLoading } = useOliveData(db, userId, isAuthReady);

  // Show error screen if initialization failed
  if (initError) {
    return (
      <View style={styles.errorContainer}>
        <Text style={styles.errorTitle}>Error Loading App</Text>
        <Text style={styles.errorText}>Code: {initError}</Text>
        <Text style={styles.errorInstructions}>
          Firebase ki files load nahi ho rahi hain. Baraye meherbani 'package.json' ko check karein aur Snack ko Reload karein.
        </Text>
      </View>
    );
  }

  // Function: Item ko cart mein daalna
  const handleAddToCart = useCallback((product) => {
    // Check karein ke item pehle se cart mein hai ya nahi
    const existingItem = cart.find(item => item.id === product.id);
    
    if (existingItem) {
      // Agar hai, to sirf quantity badhao
      setCart(cart.map(item => 
        item.id === product.id ? { ...item, quantity: item.quantity + 1 } : item
      ));
    } else {
      // Agar naya hai, to naye item ke taur par add karo
      setCart([...cart, { ...product, quantity: 1 }]);
    }
    Alert.alert("Product Added", `${product.name} Cart mein shamil ho gaya hai.`);
  }, [cart]);
  
  // Function: Jab user checkout kare
  const handleCheckout = useCallback(() => {
    const total = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);
    Alert.alert(
      "Order Placed!", 
      `Total Rs ${total} ka order confirm ho gaya hai. Jald hi aapse raabta kiya jayega.`,
      [{ text: "OK", onPress: () => setCart([]) }] // Checkout ke baad cart khali kar do
    );
    setActiveTab('Home');
  }, [cart]);


  // Count kitne items hain cart mein (Badge ke liye)
  const cartItemCount = cart.length;

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar barStyle="dark-content" backgroundColor="#f3f4f6" />
      
      {/* Screen Area */}
      <View style={styles.screenArea}>
        {activeTab === 'Home' && <HomeScreen userId={userId} />}
        {activeTab === 'Scan' && <ScanScreen />}
        {activeTab === 'Shop' && (
          <ShopScreen 
            products={products} 
            loading={productsLoading} 
            onAddToCart={handleAddToCart} 
            db={db}
            userId={userId}
            isAuthReady={isAuthReady}
          />
        )}
        {activeTab === 'Cart' && <CartScreen cart={cart} onCheckout={handleCheckout} />}
      </View>

      {/* Custom Bottom Navigation Bar */}
      <View style={styles.navBar}>
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Home')}>
          <Text style={{fontSize: 24}}>🏠</Text>
          <Text style={[styles.navText, activeTab === 'Home' && styles.activeNavText]}>Home</Text>
        </TouchableOpacity>
        
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Scan')}>
          <View style={styles.scanBtnOuter}>
            <Text style={{fontSize: 28}}>📷</Text>
          </View>
        </TouchableOpacity>
        
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Shop')}>
          <Text style={{fontSize: 24}}>🛍️</Text>
          <Text style={[styles.navText, activeTab === 'Shop' && styles.activeNavText]}>Store</Text>
        </TouchableOpacity>
        
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Cart')}>
          <Text style={{fontSize: 24}}>🛒</Text>
          <Text style={[styles.navText, activeTab === 'Cart' && styles.activeNavText]}>Cart</Text>
          {/* Cart item count dikhane ke liye */}
          {cartItemCount > 0 && (
            <View style={styles.cartBadge}>
              <Text style={styles.cartBadgeText}>{cartItemCount}</Text>
            </View>
          )}
        </TouchableOpacity>
      </View>
      {/* Debug view to show current user ID */}
      <View style={{padding: 5, backgroundColor: '#333', alignItems: 'center'}}>
        <Text style={{color: 'white', fontSize: 10}}>App ID: {appId}</Text>
      </View>
    </SafeAreaView>
  );
}
