// --- GEMINI API FUNCTION (For AI Scan) ---
// Note: apiKey will be automatically injected by the environment.
const GEMINI_API_KEY = "";
const GEMINI_API_URL = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${GEMINI_API_KEY}`;

// Function to call the Gemini API for diagnosis
const diagnoseDisease = async (imagePrompt) => {
  const systemPrompt = "Aap ek mahir Olive Crop Specialist (Zaitoon ki fasal ke maahir) hain. Aapka kaam hai di gayi photo (ya tasveer) ke description par base karke zaitoon ke pattay ki bimari ka naam, khatra (risk level: Low/Medium/High), aur ilaj (cure) ko Urdu/Hindi (Roman script) mein batana. Jawaab sirf JSON format mein dein. Agar koi bimari na ho to 'Healthy' status dein.";
  
  const userQuery = `Diye gaye zaitoon ke pattay ki bimari ki tashkhees (diagnosis) karein. Photo mein ye nazar aa raha hai: ${imagePrompt}`;

  const payload = {
    contents: [{ parts: [{ text: userQuery }] }],
    systemInstruction: { parts: [{ text: systemPrompt }] },
    generationConfig: {
      responseMimeType: "application/json",
      responseSchema: {
        type: "OBJECT",
        properties: {
          diseaseName: { type: "STRING", description: "Bimari ka naam. Agar koi nahi, toh 'Healthy'." },
          riskLevel: { type: "STRING", description: "Khatra level: Low, Medium, ya High." },
          cure: { type: "STRING", description: "Bimari ka mukammal ilaj aur spray ki sifarish." },
          icon: { type: "STRING", description: "Bimari ke liye ek appropriate emoji, ya Healthy ke liye 🌿." }
        },
        required: ["diseaseName", "riskLevel", "cure", "icon"]
      }
    }
  };

  try {
    const response = await fetch(GEMINI_API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    });

    if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
    }

    const result = await response.json();
    const jsonText = result.candidates?.[0]?.content?.parts?.[0]?.text;
    
    if (jsonText) {
      const cleanJsonText = jsonText.replace(/```json|```/g, '').trim();
      return JSON.parse(cleanJsonText);
    }
    return { diseaseName: "Unknown Error", riskLevel: "N/A", cure: "Jawab mil nahi saka.", icon: "❓" };

  } catch (error) {
    console.error("Gemini API Error:", error);
    return { diseaseName: "Connection Error", riskLevel: "High", cure: `API call failed: ${error.message}. Internet check karein.`, icon: "❌" };
  }
};

// --- 2. AI SCAN SCREEN ---
const ScanScreen = () => {
  const [scanning, setScanning] = useState(false);
  const [result, setResult] = useState(null);
  const [imageDescription, setImageDescription] = useState("Pattay par chote gol kaale daagh hain.");

  const startScan = async () => {
    if (!imageDescription.trim()) {
        return Alert.alert("Error", "Bimari ki tafseel (description) likhna zaroori hai.");
    }
    setScanning(true);
    setResult(null);
    
    const diagnosis = await diagnoseDisease(imageDescription);

    setResult(diagnosis);
    setScanning(false);
    
    if (diagnosis.diseaseName !== "Connection Error" && diagnosis.diseaseName !== "Unknown Error") {
        Alert.alert("Scan Complete", `${diagnosis.diseaseName} detect ho gayi hai!`);
    } else {
        Alert.alert("Scan Failed", diagnosis.cure);
    }
  };

  return (
    <View style={styles.centerContainer}>
      <Text style={styles.headerTitle}>AI Scanner (Gemini)</Text>
      
      {result ? (
        <View style={styles.resultCard}>
          <Text style={{fontSize: 50}}>{result.icon}</Text>
          <Text style={styles.resultTitle}>{result.diseaseName}</Text>
          <Text style={styles.matchText}>Risk: {result.riskLevel}</Text>
          <View style={styles.divider} />
          <Text style={styles.label}>Ilaj (Solution):</Text>
          <Text style={styles.cureText}>{result.cure}</Text>
          <TouchableOpacity style={styles.retryBtn} onPress={() => setResult(null)}>
            <Text style={styles.btnTextWhite}>Dobara Scan Karein</Text>
          </TouchableOpacity>
        </View>
      ) : (
        <View style={{alignItems: 'center', width: '100%'}}>
          {scanning ? (
            <View style={{alignItems: 'center'}}>
              <ActivityIndicator size="large" color="#65a30d" />
              <Text style={{marginTop: 10, color: '#65a30d', fontSize: 18}}>AI Bimari ki Tashkhees kar raha hai...</Text>
            </View>

                <Text style={styles.btnTextWhite}>AI Scan Shuru Karein</Text>
              </TouchableOpacity>
            </>

  );
};

// --- MAIN APP (Navigation & State Logic) ---
export default function App() {
  const [activeTab, setActiveTab] = useState('Home');
  const [cart, setCart] = useState([]); 
  
  const products = INITIAL_PRODUCTS;

  const handleAddToCart = useCallback((product) => {
    const existingItem = cart.find(item => item.id === product.id);
    
    if (existingItem) {
      setCart(cart.map(item => 
        item.id === product.id ? { ...item, quantity: item.quantity + 1 } : item
      ));
    } else {
      setCart([...cart, { ...product, quantity: 1 }]);
    }
    Alert.alert("Product Added", `${product.name} Cart mein shamil ho gaya hai.`);
  }, [cart]);
  
  const handleCheckout = useCallback(() => {
    const total = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);
    Alert.alert(
      "Order Placed!", 
      `Total Rs ${total} ka order confirm ho gaya hai. Jald hi aapse raabta kiya jayega.`,
      [{ text: "OK", onPress: () => setCart([]) }]
    );
    setActiveTab('Home');
  }, [cart]);

  const cartItemCount = cart.length;

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar barStyle="dark-content" backgroundColor="#f3f4f6" />
      
      {/* Screen Area */}
      <View style={styles.screenArea}>
        {activeTab === 'Home' && <HomeScreen />}
        {activeTab === 'Scan' && <ScanScreen />}
        {activeTab === 'Shop' && (
          <ShopScreen 
            products={products} 
            onAddToCart={handleAddToCart} 
          />
        )}
        {activeTab === 'Cart' && <CartScreen cart={cart} onCheckout={handleCheckout} />}
      </View>

      {/* Custom Bottom Navigation Bar */}
      <View style={styles.navBar}>
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Home')}>
          <Text style={{fontSize: 24}}>🏠</Text>
          <Text style={[styles.navText, activeTab === 'Home' && styles.activeNavText]}>Home</Text>
        </TouchableOpacity>
        
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Scan')}>
          <View style={styles.scanBtnOuter}>
            <Text style={{fontSize: 28}}>📷</Text>
          </View>
        </TouchableOpacity>
        
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Shop')}>
          <Text style={{fontSize: 24}}>🛍️</Text>
          <Text style={[styles.navText, activeTab === 'Shop' && styles.activeNavText]}>Store</Text>
        </TouchableOpacity>
        
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('Cart')}>
          <Text style={{fontSize: 24}}>🛒</Text>
          <Text style={[styles.navText, activeTab === 'Cart' && styles.activeNavText]}>Cart</Text>
          {cartItemCount > 0 && (
            <View style={styles.cartBadge}>
              <Text style={styles.cartBadgeText}>{cartItemCount}</Text>
            </View>
          )}
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
}
