/* -------------------------
   Screens
   ------------------------- */

function HomeScreen() {
  const [query, setQuery] = React.useState('');
  const fade = React.useRef(new Animated.Value(0)).current;

  React.useEffect(() => {
    Animated.timing(fade, {
      toValue: 1,
      duration: 600,
      useNativeDriver: true
    }).start();
  }, []);

  const filteredVarieties = sampleVarieties.filter(v =>
    v.name.toLowerCase().includes(query.toLowerCase()) ||
    v.origin.toLowerCase().includes(query.toLowerCase())
  );

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar style="light" />
      <Header />
      <Animated.View style={{ opacity: fade, flex: 1, width: '100%' }}>
        <View style={styles.content}>
          <SearchBar value={query} onChange={setQuery} />

          <FlatList
            data={sampleCards}
            keyExtractor={(i) => i.id}
            horizontal
            showsHorizontalScrollIndicator={false}
            style={{ marginTop: 12, maxHeight: 110 }}
            contentContainerStyle={{ paddingLeft: 20 }}
            renderItem={({ item }) => <FeatureCard item={item} />}
          />

          <Text style={styles.sectionTitle}>Popular Varieties</Text>

          <FlatList
            data={filteredVarieties}
            keyExtractor={(i) => i.id}
            renderItem={({ item }) => <VarietyCard data={item} />}
            showsVerticalScrollIndicator={false}
            contentContainerStyle={{ paddingBottom: 120 }}
          />
        </View>
      </Animated.View>
    </SafeAreaView>
  );
}

   Navigation Setup

export default function App() {

      <Tab.Navigator
        initialRouteName="Home"
        screenOptions={({ route }) => ({
          headerShown: false,

          tabBarInactiveTintColor: '#e9ffe9aa',

        <Tab.Screen
          name="Home"
          component={HomeScreen}
          options={{
            tabBarIcon: ({ color, size }) => <Ionicons name="home" size={size} color={color} />
          }}
        />
        <Tab.Screen
          name="Varieties"
          component={VarietiesScreen}
          options={{
